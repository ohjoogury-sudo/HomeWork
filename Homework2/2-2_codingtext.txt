#include <string>
#include <vector>

using namespace std;

int solution(vector<vector<int>> board, int k) {
    int answer = 0;
    for(int i = 0; i < board.size(); i++)
    {
        vector<int> tmp = board[i];
        for(int j = 0; j < tmp.size(); j++)
        {
            if (i + j <= k)
            {
                answer += tmp[j];
            }
        }
    }
        
    return answer;
}
